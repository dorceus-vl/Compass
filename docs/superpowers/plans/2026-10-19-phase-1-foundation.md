# Compass Phase 1 (Foundation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A signed-in, installable Compass web app at compass.knocturnlabs.com with projects, milestones, tasks across three horizons, Today, Board, List, Upcoming, a basic Timeline, a command palette with quick add, Trash, Settings, and an offline store that syncs.

**Architecture:** One repo. `worker/` is a Hono app on Cloudflare Workers with D1 (sessions, projects, milestones, tasks, events, settings, audit) and Knocturn ID sign-in over a service binding, copied from Health and Aura. `web/` is a React + Vite PWA served as Worker static assets (`/api/*` runs the Worker first), with an IndexedDB mirror and a write queue so it works offline. Work rows are encrypted at rest with Health's crypto-box.

**Tech Stack:** TypeScript 5, Hono 4, Cloudflare Workers + D1 + Wrangler 4, React 19 + Vite 6, plain CSS with design tokens, Node 22 for self-checks (`node --experimental-strip-types`).

**Spec:** `docs/superpowers/specs/2026-10-19-compass-design.md` (sections 1–4, 6, 8, 12, 13 step 1, 14). Feature catalogue: `docs/compass-features.md`. Previews: `docs/previews.html`.

- Domain `compass.knocturnlabs.com`; Knocturn ID app id `compass`; Worker name `compass`; D1 `compass-db` (binding `DB`); R2 `compass-files` (binding `FILES`, created now, used in Phase 3).

- Nothing is hard-deleted from the API: `DELETE` sets `deleted_at`; Trash purges after 30 days by cron. Emptying Trash needs the passkey step-up.
- Work rows (`kind = 'work'`) have their text columns AES-GCM encrypted with `COMPASS_DATA_KEY` (crypto-box v2 with AAD `table.column`); personal rows are plain.

- Self-checks: `scripts/*.test.ts` run by `node --experimental-strip-types`, `assert`-based, no framework. `npm run typecheck` must pass before every commit.
- Dependencies allowed in Phase 1: `hono`, `react`, `react-dom`, dev: `vite`, `@vitejs/plugin-react`, `typescript`, `wrangler`, `@cloudflare/workers-types`, `@types/react`, `@types/react-dom`. Nothing else.

- [ ] **Step 4: worker/lib/session.ts**

Health's `session.ts` with three changes: no profiles (`c.env.DB` instead of `mainDb(c.env)`), ISO timestamps, `COMPASS_DATA_KEY`, cookie `cp_session`. Write it in full:

```ts
import { getCookie, setCookie, deleteCookie } from 'hono/cookie';
import type { Context } from 'hono';
import type { Env, SessionRow, Variables } from '../types.ts';
import { encryptString, decryptString } from './crypto-box.js';
import { idFetch, type IdTokens } from './knocturn-id.ts';
import { subOf, userAllowed } from './access.ts';

export const SESSION_COOKIE = 'cp_session';
const IDLE_TIMEOUT_MS = 60 * 60 * 1000;
const ACTIVE_LIFETIME_MS = 30 * 24 * 60 * 60 * 1000;
const ID_CHECK_MS = 60 * 60 * 1000;
type Ctx = Context<{ Bindings: Env; Variables: Variables }>;
const now = () => new Date().toISOString();

function randomToken(): string {
  const bytes = new Uint8Array(32);
  crypto.getRandomValues(bytes);
  return [...bytes].map((b) => b.toString(16).padStart(2, '0')).join('');
}
async function sha256Hex(input: string): Promise<string> {
  const digest = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(input));
  return [...new Uint8Array(digest)].map((b) => b.toString(16).padStart(2, '0')).join('');
}
const isSecure = (c: Ctx) => new URL(c.req.url).protocol === 'https:';
function setSessionCookie(c: Ctx, token: string, maxAgeMs: number) {
  setCookie(c, SESSION_COOKIE, token, { httpOnly: true, secure: isSecure(c), sameSite: 'Lax', path: '/', maxAge: Math.floor(maxAgeMs / 1000) });
}
export function clearSessionCookie(c: Ctx) {
  deleteCookie(c, SESSION_COOKIE, { path: '/' });
}

export async function getSession(c: Ctx): Promise<SessionRow | null> {
  const token = getCookie(c, SESSION_COOKIE);
  if (!token) return null;
  const row = await c.env.DB.prepare('SELECT * FROM auth_sessions WHERE token_hash = ?1').bind(await sha256Hex(token)).first<SessionRow>();
  if (!row || row.revoked) return null;
  if (new Date(row.expires_at).getTime() < Date.now()) return null;
  if (Date.now() - new Date(row.last_seen_at).getTime() > IDLE_TIMEOUT_MS) return null;
  return row;
}

const encryptTokens = (c: Ctx, t: IdTokens) =>
  encryptString(JSON.stringify({ access_token: t.access_token, refresh_token: t.refresh_token }), c.env.COMPASS_DATA_KEY, 'auth_sessions.id_tokens_enc');

/** tokens = null makes a dev session with no Knocturn ID behind it (dev-login only). */
export async function createSession(c: Ctx, tokens: IdTokens | null): Promise<void> {
  const token = randomToken();
  const t = now();
  await c.env.DB.prepare(
    `INSERT INTO auth_sessions (id, token_hash, created_at, last_seen_at, expires_at, user_agent, ip, revoked, id_tokens_enc, id_checked_at)
     VALUES (?1, ?2, ?3, ?3, ?4, ?5, ?6, 0, ?7, ?3)`,
  )
    .bind(crypto.randomUUID(), await sha256Hex(token), t, new Date(Date.now() + ACTIVE_LIFETIME_MS).toISOString(),
      c.req.header('user-agent') ?? null, c.req.header('cf-connecting-ip') ?? null, tokens ? await encryptTokens(c, tokens) : null)
    .run();
  setSessionCookie(c, token, ACTIVE_LIFETIME_MS);
}

export async function idTokens(c: Ctx, session: SessionRow): Promise<IdTokens | null> {
  if (!session.id_tokens_enc) return null;
  return JSON.parse(await decryptString(session.id_tokens_enc, c.env.COMPASS_DATA_KEY, 'auth_sessions.id_tokens_enc'));
}

/** Hourly: refresh the Knocturn ID tokens; a refusal ends the session here, an outage doesn't (INTEGRATION.md §3). */
export async function checkWithId(c: Ctx, session: SessionRow): Promise<boolean> {
  if (session.id_checked_at && Date.now() - new Date(session.id_checked_at).getTime() < ID_CHECK_MS) return true;
  const claimed = await c.env.DB.prepare('UPDATE auth_sessions SET id_checked_at = ?1 WHERE id = ?2 AND id_checked_at IS ?3')
    .bind(now(), session.id, session.id_checked_at).run();
  if (!claimed.meta.changes) return true;
  const tokens = await idTokens(c, session);
  if (!tokens) return true;
  const r = await idFetch(c, '/v1/auth/refresh', { refresh_token: tokens.refresh_token }).catch(() => null);
  if (r?.ok) {
    const fresh = await r.json<IdTokens>();
    if (!userAllowed(c.env.COMPASS_USERS, subOf(fresh.access_token))) { await revokeSession(c.env.DB, session.id); return false; }
    await c.env.DB.prepare('UPDATE auth_sessions SET id_tokens_enc = ?1 WHERE id = ?2').bind(await encryptTokens(c, fresh), session.id).run();
    return true;
  }
  if (r?.status === 401) { await revokeSession(c.env.DB, session.id); return false; }
  return true;
}

export async function touchSession(c: Ctx, sessionId: string) {
  await c.env.DB.prepare('UPDATE auth_sessions SET last_seen_at = ?1 WHERE id = ?2').bind(now(), sessionId).run();
  const token = getCookie(c, SESSION_COOKIE);
  if (token) setSessionCookie(c, token, ACTIVE_LIFETIME_MS);
}
export async function revokeSession(db: D1Database, sessionId: string) {
  await db.prepare('UPDATE auth_sessions SET revoked = 1 WHERE id = ?1').bind(sessionId).run();
}
export async function endSession(c: Ctx, session: SessionRow) {
  const tokens = await idTokens(c, session).catch(() => null);
  if (tokens) await idFetch(c, '/v1/auth/logout', { refresh_token: tokens.refresh_token }).catch(() => null);
  await revokeSession(c.env.DB, session.id);
}
```

```ts
import type { Env } from '../types.ts';
import { encryptString, decryptString } from './crypto-box.js';

// Which text columns are ciphertext on a work row. AAD binds each value to its column (table.column).
export const ENC_COLS = {
  projects: ['name', 'rag_reason', 'sponsor', 'objectives', 'scope_in', 'scope_out', 'success_criteria'],
  milestones: ['name'],
  tasks: ['title', 'notes', 'waiting_on'],
  task_events: ['data'],
} as const;
export type EncTable = keyof typeof ENC_COLS;
type Row = Record<string, unknown> & { enc_v?: number; kind?: string };

/** Encrypts the table's text columns when the row (or the given kind) is work. Mutates and returns the row. */
export async function seal<T extends Row>(env: Env, table: EncTable, row: T, kind = row.kind): Promise<T> {
  if (kind !== 'work') return row;
  for (const col of ENC_COLS[table]) {
    const v = row[col];
    if (typeof v === 'string' && v !== '') (row as Row)[col] = await encryptString(v, env.COMPASS_DATA_KEY, `${table}.${col}`);
  }
  row.enc_v = 2;
  return row;
}

/** Decrypts a row written by seal(); a plain row passes through. */
export async function open<T extends Row>(env: Env, table: EncTable, row: T): Promise<T> {
  if (row.enc_v !== 2) return row;
  for (const col of ENC_COLS[table]) {
    const v = row[col];
    if (typeof v === 'string' && v !== '') (row as Row)[col] = await decryptString(v, env.COMPASS_DATA_KEY, `${table}.${col}`);
  }
  return row;
}

export const openAll = <T extends Row>(env: Env, table: EncTable, rows: T[]) => Promise.all(rows.map((r) => open(env, table, r)));
```

```ts
// Recurring tasks: the next due date after one is done. Dates are YYYY-MM-DD; weekdays 1 = Monday ... 7 = Sunday.
export interface RecurRule { freq: 'daily' | 'weekly' | 'monthly'; interval: number; weekdays?: number[]; after_completion?: boolean }

export function parseRecur(json: string | null): RecurRule | null {
  if (!json) return null;
  try {
    const r = JSON.parse(json);
    if (!['daily', 'weekly', 'monthly'].includes(r?.freq)) return null;
    const interval = Number.isInteger(r.interval) && r.interval >= 1 ? r.interval : 1;
    const weekdays = Array.isArray(r.weekdays) ? r.weekdays.filter((d: unknown) => Number.isInteger(d) && (d as number) >= 1 && (d as number) <= 7) : undefined;
    return { freq: r.freq, interval, ...(weekdays?.length ? { weekdays } : {}), ...(r.after_completion ? { after_completion: true } : {}) };
  } catch { return null; }
}

const day = (iso: string) => new Date(iso.slice(0, 10) + 'T00:00:00Z');
const ymd = (d: Date) => d.toISOString().slice(0, 10);
const addDays = (d: Date, n: number) => new Date(d.getTime() + n * 86_400_000);
const isoWeekday = (d: Date) => ((d.getUTCDay() + 6) % 7) + 1;

function addMonthsClamped(d: Date, n: number, dayOfMonth: number): Date {
  const first = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth() + n, 1));
  const last = new Date(Date.UTC(first.getUTCFullYear(), first.getUTCMonth() + 1, 0)).getUTCDate();
  return new Date(Date.UTC(first.getUTCFullYear(), first.getUTCMonth(), Math.min(dayOfMonth, last)));
}

export function nextOccurrence(dueAt: string, rule: RecurRule, doneAt: string): string | null {
  const done = day(doneAt);
  const base = rule.after_completion ? done : day(dueAt);
  let next: Date;
  if (rule.freq === 'daily') next = addDays(base, rule.interval);
  else if (rule.freq === 'monthly') next = addMonthsClamped(base, rule.interval, day(dueAt).getUTCDate());
  else if (rule.weekdays?.length) {
    const days = [...rule.weekdays].sort((a, b) => a - b);
    const wd = isoWeekday(base);
    const later = days.find((d) => d > wd);
    next = later ? addDays(base, later - wd) : addDays(base, 7 * rule.interval - wd + days[0]);
  } else next = addDays(base, 7 * rule.interval);
  // Never schedule into the past: an overdue task done today recurs from today.
  while (next.getTime() <= done.getTime()) next = rule.freq === 'daily' ? addDays(next, rule.interval) : rule.freq === 'monthly' ? addMonthsClamped(next, rule.interval, day(dueAt).getUTCDate()) : addDays(next, 7 * rule.interval);
  return ymd(next);
}
```

```ts
// Fractional ordering inside a board column: a move writes one row, not the whole column.
// ponytail: positions can get very close after many moves; renumber the column if a gap drops under 1e-6.
export function positionBetween(prev: number | null, next: number | null): number {
  if (prev === null && next === null) return 1000;
  if (prev === null) return next! - 1000;
  if (next === null) return prev + 1000;
  return (prev + next) / 2;
}
```

```ts
import type { Context } from 'hono';
import type { Env, Variables } from '../types.ts';
import { idFetch } from './knocturn-id.ts';
import { idTokens } from './session.ts';

type Ctx = Context<{ Bindings: Env; Variables: Variables }>;

/** navigator.credentials.get() options from Knocturn ID (INTEGRATION.md §8b step-up). */
export async function stepUpChallenge(c: Ctx): Promise<Response> {
  const tokens = await idTokens(c, c.get('session')).catch(() => null);
  if (!tokens) return c.json({ options: null, dev: true });
  const r = await idFetch(c, '/v1/passkeys/challenge', {}, tokens.access_token).catch(() => null);
  if (!r?.ok) return c.json({ error: r?.status === 403 ? 'Add a passkey to your Knocturn ID first.' : "Can't reach Knocturn ID right now." }, r?.status === 403 ? 403 : 503);
  return c.json(await r.json());
}

/** null when the request carries an accepted passkey answer in X-Step-Up (JSON), else the 403 to send. */
export async function requireStepUp(c: Ctx): Promise<Response | null> {
  const tokens = await idTokens(c, c.get('session')).catch(() => null);
  if (!tokens) return null; // dev session (wrangler dev only)
  const raw = c.req.header('x-step-up');
  if (!raw) return c.json({ error: 'Confirm with your passkey to continue.', step_up: true }, 403);
  let credential: unknown;
  try { credential = JSON.parse(raw); } catch { return c.json({ error: 'Bad step-up answer.', step_up: true }, 403); }
  const r = await idFetch(c, '/v1/passkeys/verify', { credential }, tokens.access_token).catch(() => null);
  if (!r) return c.json({ error: "Can't reach Knocturn ID right now." }, 503);
  if (r.status === 429) return c.json({ error: 'Too many attempts. Wait 15 minutes.' }, 429);
  if (r.status !== 204) return c.json({ error: "That passkey wasn't accepted. Try again.", step_up: true }, 403);
  return null;
}
```

```ts
/** Cron: rows in Trash for 30 days are deleted (all = everything in Trash, for Empty Trash). */
export async function purgeTrash(env: Env, all = false): Promise<Record<string, number>> {
  const cutoff = all ? '9999' : daysAgo(TRASH_DAYS);
  const out: Record<string, number> = {};
  const ev = await env.DB.prepare('DELETE FROM task_events WHERE task_id IN (SELECT id FROM tasks WHERE deleted_at IS NOT NULL AND deleted_at < ?1)').bind(cutoff).run();
  out.task_events = ev.meta.changes;
  for (const table of ['tasks', 'milestones', 'projects'] as const) {
    const r = await env.DB.prepare(`DELETE FROM ${table} WHERE deleted_at IS NOT NULL AND deleted_at < ?1`).bind(cutoff).run();
    out[table] = r.meta.changes;
  }
  return out;
}
export async function purgeAudit(env: Env): Promise<void> {
  await env.DB.prepare('DELETE FROM audit_log WHERE at < ?1').bind(daysAgo(AUDIT_DAYS)).run();
}
```

```ts
// Single user, last write wins by updated_at (spec §8). Pulls are incremental, so a row missing remotely is kept.
export function mergeRows<T extends { id: string; updated_at: string }>(local: T[], remote: T[]): T[] {
  const out = new Map(local.map((r) => [r.id, r]));
  for (const r of remote) { const l = out.get(r.id); if (!l || r.updated_at >= l.updated_at) out.set(r.id, r); }
  return [...out.values()];
}
```

```ts
import { api, SignedOut, online } from '../api.ts';
import { idb } from './db.ts';
import { mergeRows } from './merge.ts';
import type { Task, Project, Milestone } from '../../../shared/types.ts';

/** A queued write. `tempId` maps an optimistic create to the server's id when the reply comes back. */
export interface QueuedOp { seq?: number; method: 'POST' | 'PATCH' | 'DELETE'; path: string; json?: unknown; tempId?: string; store?: 'tasks' | 'projects' | 'milestones' }
const listeners = new Set<() => void>();
export const onChange = (l: () => void) => { listeners.add(l); return () => { listeners.delete(l); }; };
export const notify = () => listeners.forEach((l) => l());
let replaying = false;

export async function pull(): Promise<void> {
  const meta = (await idb.get<{ id: string; at: string }>('meta', 'last_pull'))?.at;
  const [t, p] = await Promise.all([
    api<{ tasks: Task[] }>(`/tasks?status=all${meta ? `&since=${encodeURIComponent(meta)}` : ''}`),
    api<{ projects: Project[] }>('/projects'),
  ]);
  const localTasks = await idb.all<Task>('tasks');
  await idb.putAll('tasks', mergeRows(localTasks, t.tasks));
  await idb.putAll('projects', mergeRows(await idb.all<Project>('projects'), p.projects));
  const ms = await Promise.all(p.projects.map((pr) => api<{ milestones: Milestone[] }>(`/projects/${pr.id}`).then((r) => r.milestones)));
  await idb.clear('milestones'); await idb.putAll('milestones', ms.flat());
  await idb.put('meta', { id: 'last_pull', at: new Date().toISOString() });
  notify();
}

export async function enqueue(op: QueuedOp): Promise<void> { await idb.put('queue', op); if (online()) void replay(); }

/** Sends queued writes in order; stops at the first network failure and retries when back online. */
export async function replay(onSignedOut?: () => void): Promise<void> {
  if (replaying) return; replaying = true;
  try {
    const ops = (await idb.all<QueuedOp>('queue')).sort((a, b) => a.seq! - b.seq!);
    const idMap = new Map<string, string>();
    for (const op of ops) {
      const path = op.path.replace(/tmp-[0-9a-f-]+/g, (m) => idMap.get(m) ?? m);
      try {
        const r = await api<Record<string, { id: string } | undefined>>(path, { method: op.method, json: op.json });
        if (op.tempId && op.store) {
          const real = (r.task ?? r.project ?? r.milestone)!;
          idMap.set(op.tempId, real.id);
          await idb.del(op.store, op.tempId); await idb.put(op.store, real);
        }
        await idb.del('queue', op.seq!);
      } catch (e) {
        if (e instanceof SignedOut) { onSignedOut?.(); return; }
        if (e instanceof TypeError) return; // offline: keep the op, try later
        await idb.del('queue', op.seq!); // 4xx: the server refused it; drop it rather than block the queue
        console.error('dropped queued write', op, e);
      }
    }
  } finally { replaying = false; notify(); }
}

export function startSync(onSignedOut: () => void): () => void {
  const run = () => { if (!online()) return; replay(onSignedOut).then(pull).catch((e) => { if (e instanceof SignedOut) onSignedOut(); }); };
  run();
  window.addEventListener('online', run);
  const timer = setInterval(run, 60_000);
  return () => { window.removeEventListener('online', run); clearInterval(timer); };
}
```

```ts
export const today = () => new Intl.DateTimeFormat('en-CA', { year: 'numeric', month: '2-digit', day: '2-digit' }).format(new Date());
export const addDays = (ymd: string, n: number) => new Date(new Date(ymd + 'T00:00:00').getTime() + n * 86_400_000).toLocaleDateString('en-CA');
export const startOfWeek = (ymd: string) => addDays(ymd, -((new Date(ymd + 'T00:00:00').getDay() + 6) % 7));
const DOW = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
export function fmtDue(ymd: string | null, t = today()): { text: string; tone: 'over' | 'due' | '' } {
  if (!ymd) return { text: '', tone: '' };
  const diff = Math.round((new Date(ymd + 'T00:00:00').getTime() - new Date(t + 'T00:00:00').getTime()) / 86_400_000);
  if (diff < 0) return { text: diff === -1 ? 'Yesterday' : `${-diff}d late`, tone: 'over' };
  if (diff === 0) return { text: 'Today', tone: 'due' };
  if (diff === 1) return { text: 'Tomorrow', tone: 'due' };
  if (diff < 7) return { text: DOW[new Date(ymd + 'T00:00:00').getDay()], tone: '' };
  return { text: new Date(ymd + 'T00:00:00').toLocaleDateString('en-CA', { month: 'short', day: 'numeric' }), tone: '' };
}
```

```ts
import type { Task, Project } from '../../shared/types.ts';

export interface Query { words: string[]; status?: string; kind?: string; project?: string; horizon?: string; due?: string; tag?: string }

export function parseQuery(q: string): Query {
  const out: Query = { words: [] };
  for (const tok of q.trim().split(/\s+/).filter(Boolean)) {
    const m = tok.match(/^(status|kind|project|horizon|due|tag):(.+)$/i);
    if (m) (out as Record<string, unknown>)[m[1].toLowerCase()] = m[2].toLowerCase(); else out.words.push(tok.toLowerCase());
  }
  return out;
}

export function applyQuery(tasks: Task[], q: Query, projects: Pick<Project, 'id' | 'name'>[], today: string): Task[] {
  const pid = q.project ? projects.find((p) => p.name.toLowerCase().startsWith(q.project!))?.id ?? '__none__' : null;
  return tasks.filter((t) => {
    if (t.deleted_at) return false;
    if (q.status === 'all') { /* any */ } else if (q.status) { if (t.status !== q.status) return false; } else if (t.status === 'done') return false;
    if (q.kind && t.kind !== q.kind) return false;
    if (pid && t.project_id !== pid) return false;
    if (q.horizon && t.horizon !== q.horizon) return false;
    if (q.tag && !JSON.parse(t.tags || '[]').map((x: string) => x.toLowerCase()).includes(q.tag)) return false;
    if (q.due) {
      if (q.due === 'overdue') { if (!t.due_at || t.due_at >= today) return false; }
      else if (q.due === 'none') { if (t.due_at) return false; }
      else { const m = q.due.match(/^([<>]=?)?(\d{4}-\d{2}-\d{2})$/); if (!m || !t.due_at) return false; const op = m[1] ?? '='; const d = m[2];
        if (op === '<' ? !(t.due_at < d) : op === '<=' ? !(t.due_at <= d) : op === '>' ? !(t.due_at > d) : op === '>=' ? !(t.due_at >= d) : t.due_at !== d) return false; }
    }
    return q.words.every((w) => t.title.toLowerCase().includes(w));
  });
}
```

```ts
import type { Task } from '../../shared/types.ts';
export const dayIndex = (ymd: string, from: string) => Math.round((Date.parse(ymd + 'T00:00:00Z') - Date.parse(from + 'T00:00:00Z')) / 86_400_000);
const add = (ymd: string, n: number) => new Date(Date.parse(ymd + 'T00:00:00Z') + n * 86_400_000).toISOString().slice(0, 10);
export interface Row { id: string; title: string; x: number; w: number; y: number; due: boolean }
export function layout(tasks: Pick<Task, 'id' | 'title' | 'start_at' | 'due_at'>[], from: string, pxPerDay: number, rowH = 34): Row[] {
  return tasks.filter((t) => t.due_at).map((t, i) => {
    const s = dayIndex(t.start_at ?? t.due_at!, from), e = dayIndex(t.due_at!, from);
    return { id: t.id, title: t.title, x: s * pxPerDay, w: Math.max(1, e - s + 1) * pxPerDay, y: i * rowH, due: !t.start_at };
  });
}
```

```ts
// Quick-add grammar (spec §6): title [#project] [@work|@personal] [!1-3] [+tag] [date words]
// Date words: today, tomorrow, mon…sun, next week, in N days, oct 24, 24/10, every day|week|month.
import type { Project } from '../../shared/types.ts';
const DOW = ['sun', 'mon', 'tue', 'wed', 'thu', 'fri', 'sat'];
const MON = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
const add = (ymd: string, n: number) => new Date(Date.parse(ymd + 'T00:00:00Z') + n * 86_400_000).toISOString().slice(0, 10);
const pad = (n: number) => String(n).padStart(2, '0');

export function parseQuickAdd(text: string, projects: Pick<Project, 'id' | 'name' | 'kind'>[], today: string) {
  let s = ' ' + text.trim() + ' ';
  const out = { title: '', project_id: null as string | null, kind: 'personal' as 'personal' | 'work', priority: 0, due_at: null as string | null, recur: null as string | null, tags: [] as string[] };
  const take = (re: RegExp, f: (m: RegExpMatchArray) => void) => { const m = s.match(re); if (m) { f(m); s = s.replace(m[0], ' '); } };
  take(/ #(\S+)/, (m) => { const p = projects.find((x) => x.name.toLowerCase().startsWith(m[1].toLowerCase())); if (p) { out.project_id = p.id; out.kind = p.kind; } });
  take(/ @(work|personal)/i, (m) => { if (!out.project_id) out.kind = m[1].toLowerCase() as 'work'; });
  take(/ !([1-3])/, (m) => { out.priority = Number(m[1]); });
  for (;;) { const m = s.match(/ \+(\S+)/); if (!m) break; out.tags.push(m[1]); s = s.replace(m[0], ' '); }
  take(/ every (day|week|month|weekday)s?/i, (m) => { const w = m[1].toLowerCase(); out.recur = w === 'weekday' ? '{"freq":"weekly","interval":1,"weekdays":[1,2,3,4,5]}' : `{"freq":"${w === 'day' ? 'daily' : w + 'ly'}","interval":1}`; });
  const y = Number(today.slice(0, 4));
  take(/ today /i, () => { out.due_at = today; });
  take(/ tomorrow /i, () => { out.due_at = add(today, 1); });
  take(/ next week /i, () => { out.due_at = add(today, 7 - ((new Date(today + 'T00:00:00Z').getUTCDay() + 6) % 7)); });
  take(/ in (\d+) days? /i, (m) => { out.due_at = add(today, Number(m[1])); });
  take(/ (\d{1,2})\/(\d{1,2}) /, (m) => { out.due_at = `${y}-${pad(Number(m[2]))}-${pad(Number(m[1]))}`; });
  take(new RegExp(` (${MON.join('|')})[a-z]* (\\d{1,2}) `, 'i'), (m) => { out.due_at = `${y}-${pad(MON.indexOf(m[1].toLowerCase().slice(0, 3)) + 1)}-${pad(Number(m[2]))}`; });
  take(new RegExp(` (${DOW.join('|')})[a-z]* `, 'i'), (m) => { const want = DOW.indexOf(m[1].toLowerCase().slice(0, 3)); const cur = new Date(today + 'T00:00:00Z').getUTCDay(); out.due_at = add(today, ((want - cur + 7) % 7) || 7); });
  if (out.due_at && out.due_at < today && !/\d{1,2}\/|[a-z]{3} \d/.test(text)) out.due_at = today;
  out.title = s.replace(/\s+/g, ' ').trim();
  return out;
}
```

```tsx
import { useEffect, useRef, useState } from 'react';
```

```bash
#!/usr/bin/env bash
# End-to-end against a local wrangler dev (DEV_LOGIN=1 in .dev.vars). Signs in, makes a project, a task, moves it, trashes and restores it.
set -euo pipefail
B=${BASE:-http://localhost:8790}; J=/tmp/compass-smoke.jar; rm -f $J
H=(-s -b $J -c $J -H "Origin: $B" -H "Content-Type: application/json")
fail() { echo "smoke: FAIL $1"; exit 1; }
curl "${H[@]}" -X POST $B/api/auth/dev-login | grep -q '"ok":true' || fail dev-login
P=$(curl "${H[@]}" -X POST $B/api/projects -d '{"name":"Smoke project","kind":"work"}' | node -pe 'JSON.parse(require("fs").readFileSync(0)).project.id')
T=$(curl "${H[@]}" -X POST $B/api/tasks -d "{\"title\":\"Smoke task\",\"project_id\":\"$P\",\"due_at\":\"2026-01-01\"}" | node -pe 'JSON.parse(require("fs").readFileSync(0)).task.id')
curl "${H[@]}" $B/api/today | grep -q 'Smoke task' || fail today
curl "${H[@]}" -X POST $B/api/tasks/$T/move -d '{"status":"progress"}' | grep -q '"status":"progress"' || fail move
curl "${H[@]}" $B/api/board | grep -q '"progress":\[{' || fail board
curl "${H[@]}" -X DELETE $B/api/tasks/$T | grep -q '"ok":true' || fail trash
curl "${H[@]}" -X POST $B/api/trash/tasks/$T/restore | grep -q '"ok":true' || fail restore
curl "${H[@]}" -X DELETE $B/api/projects/$P >/dev/null; curl "${H[@]}" -X DELETE $B/api/tasks/$T >/dev/null
curl "${H[@]}" -X POST $B/api/trash/empty | grep -q purged || fail empty
curl -s -o /dev/null -w '%{http_code}' $B/api/tasks | grep -q 401 || fail "guard (no cookie)"
echo "smoke: ok"
```
